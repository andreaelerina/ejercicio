Deforestación (Hansen GFC + alertas RADD / GLAD-L) * 2. Cambios en la cobertura detectados con radar (Sentinel-1, atraviesa nubes) * 3. Minería aluvial ilegal (MapBiomas Colombia, clase 30) * 4. Para uso institucional, reemplácelos por los polígonos * oficiales (RUNAP / ICANH / Parques Nacionales) subidos como Asset. * • Verifique en el Catálogo de GEE las rutas de los assets de CONFIG: las * versiones se actualizan (Hansen, MapBiomas, RADD). * • Es una herramienta de ALERTA TEMPRANA: toda alerta debe verificarse con * imágenes de alta resolución y trabajo de campo antes de actuar. *****************************************************************************/ // ============================== PASO 1 · CONFIGURACIÓN ====================== var CONFIG = { HANSEN: 'UMD/hansen/global_forest_change_2023_v1_11', // actualizar si hay versión nueva S1: 'COPERNICUS/S1_GRD', RADD: 'projects/radar-wur/raddalert/v1', GLAD: 'projects/glad/alert/UpdResult', MAPBIOMAS: 'projects/mapbiomas-public/assets/colombia/collection1/mapbiomas_colombia_collection1_integration_v1', MB_ANIO: 2022, // año de la clasificación MapBiomas VIAS: 'projects/sat-io/open-datasets/GRIP4/South-America', DIAS_VENTANA: 90 // ventana de análisis por defecto }; // Sitios piloto: coordenadas APROXIMADAS (lon, lat) y radio de análisis (m) var SITIOS = { 'San Agustín (Huila)': {lon: -76.284, lat: 1.886, radio: 12000}, 'Tierradentro (Cauca)': {lon: -76.040, lat: 2.570, radio: 10000}, 'Teyuna–Ciudad Perdida (Magdalena)': {lon: -73.925, lat: 11.038, radio: 12000}, 'Chiribiquete – arte rupestre': {lon: -72.700, lat: 0.200, radio: 25000} }; // Escalas de color (paletas) ------------------------------------------------ var PAL = { // Año de pérdida Hansen: amarillo (antiguo) → rojo oscuro (reciente) hansen: ['FFFFB2', 'FED976', 'FEB24C', 'FD8D3C', 'F03B20', 'BD0026'], // Fecha de alerta RADD: amarillo (inicio de ventana) → naranja oscuro (reciente) radd: ['FEE391', 'FE9929', 'CC4C02'], // Cambio Sentinel-1 VH (dB): rojo = baja retrodispersión (posible pérdida), // azul = aumento (posible construcción / suelo expuesto rugoso) s1: ['B2182B', 'EF8A62', 'FDDBC7', 'F7F7F7', 'D1E5F0', '67A9CF', '2166AC'], // Índice de presión 1–5: de amarillo claro a granate indice: ['FFFFB2', 'FECC5C', 'FD8D3C', 'F03B20', 'BD0026'], glad: ['FF00FF'], vias: ['FFFFFF'] }; // Clases MapBiomas Colombia (id, nombre, color) var MB = { ids: [3, 4, 5, 6, 11, 12, 13, 15, 18, 21, 24, 25, 30, 33], nombres: ['Formación forestal', 'Formación sabánica', 'Manglar', 'Bosque inundable', 'Humedal', 'Formación herbácea', 'Otra formación no forestal', 'Pasto', 'Agricultura', 'Mosaico agropecuario', 'Infraestructura urbana', 'Otra área sin vegetación', 'Minería', 'Río, lago u océano'], colores: ['1F8D49', '7DC975', '04381D', '026975', '519799', 'D6BC74', 'D89F5C', 'FFD966', 'E974ED', 'FFEFC3', 'D4271E', 'DB4D4F', '9C0027', '0000FF'] }; // Umbrales del semáforo (porcentaje del área del sitio) var SEMAFORO = { altoIdx2: 1, // % del área con ≥2 amenazas coincidentes → ALTO medioIdx1: 5 // % del área con ≥1 amenaza → MEDIO }; // Datasets base -------------------------------------------------------------- var gfc = ee.Image(CONFIG.HANSEN); var tc2000 = gfc.select('treecover2000'); var lossyear = gfc.select('lossyear'); // RADD: última imagen de alertas para Sudamérica ('sa'); bandas: Alert, Date (YYDOY) var radd = ee.Image( ee.ImageCollection(CONFIG.RADD) .filterMetadata('layer', 'contains', 'alert') .filterMetadata('geography', 'equals', 'sa') .sort('system:time_end', false) .first()); // ============================== PASO 2 · FUNCIONES DE ANÁLISIS ============== // --- Utilidades de fecha (cliente) --- function fmtFecha(d) { return d.toISOString().slice(0, 10); } // Convierte 'YYYY-MM-DD' a código YYDOY (formato de la banda Date de RADD) function codigoYYDOY(txt) { var d = new Date(txt + 'T00:00:00Z'); var y = d.getUTCFullYear(); var doy = Math.floor((d.getTime() - Date.UTC(y, 0, 1)) / 86400000) + 1; return (y - 2000) * 1000 + doy; } // Crea el feature (círculo) de un sitio function feat(nombre) { var s = SITIOS[nombre]; return ee.Feature(ee.Geometry.Point([s.lon, s.lat]).buffer(s.radio), {sitio: nombre}); } // Colección Sentinel-1 (IW, VH) filtrada al área function coleccionS1(geom) { return ee.ImageCollection(CONFIG.S1) .filterBounds(geom) .filter(ee.Filter.eq('instrumentMode', 'IW')) .filter(ee.Filter.listContains('transmitterReceiverPolarisation', 'VH')) .select('VH'); } // Cambio de retrodispersión VH (dB): periodo reciente MENOS misma época del año anterior // (comparar la misma época reduce el efecto de la estacionalidad). function s1Cambio(geom, ini, fin) { var col = coleccionS1(geom); var reciente = col.filterDate(ini, fin).median(); var base = col.filterDate(ee.Date(ini).advance(-1, 'year'), ee.Date(fin).advance(-1, 'year')).median(); return reciente.subtract(base).focalMedian(30, 'circle', 'meters').rename('dVH'); } // Capa GLAD-L (alertas Landsat) del año de la fecha final; conf ≥2 = probable/confirmada function gladCapa(p) { var yy = ('' + p.fin).slice(2, 4); var g = ee.ImageCollection(CONFIG.GLAD).mosaic(); var existe = g.bandNames().contains('conf' + yy); return ee.Image(ee.Algorithms.If(existe, g.select('conf' + yy).gte(2).selfMask().rename('glad'), ee.Image(0).selfMask().rename('glad'))); } // Capas binarias de amenaza + índice de presión function capas(geom, p) { var bosque = tc2000.gte(p.tc); // 1. Deforestación Hansen (pérdida desde el año elegido, sobre bosque base) var hansenR = lossyear.gte(p.anio - 2000).and(bosque).selfMask().rename('hansen'); // 2. Alertas RADD dentro de la ventana de análisis var fechaR = radd.select('Date'); var raddR = radd.select('Alert').gte(1) .and(fechaR.gte(codigoYYDOY(p.ini))) .and(fechaR.lte(codigoYYDOY(p.fin))) .selfMask().rename('radd'); // 3. Cambio radar Sentinel-1 (caída de VH sobre bosque) var dif = s1Cambio(geom, p.ini, p.fin); var s1R = dif.lt(-p.umbral).and(bosque).selfMask().rename('s1'); // 4. MapBiomas: minería (30) y urbano (24) var mb = ee.Image(CONFIG.MAPBIOMAS).select('classification_' + CONFIG.MB_ANIO); var minR = mb.eq(30).selfMask().rename('mineria'); var urbR = mb.eq(24).selfMask().rename('urbano'); // PASO 3 · Índice de presión: nº de amenazas coincidentes (0–5) var idx = hansenR.unmask(0, false) .add(raddR.unmask(0, false)) .add(s1R.unmask(0, false)) .add(minR.unmask(0, false)) .add(urbR.unmask(0, false)) .rename('indice'); return {hansen: hansenR, radd: raddR, dif: dif, s1: s1R, mb: mb, mineria: minR, urbano: urbR, idx: idx}; } // ============================== PASO 4 · ESTADÍSTICAS ======================= // Área (ha) de cada amenaza dentro del sitio function statsSitio(f, p) { var geom = f.geometry(); var L = capas(geom, p); var ha = ee.Image.pixelArea().divide(10000); var img = ee.Image.cat([ ha.rename('total'), ha.updateMask(L.hansen).rename('hansen'), ha.updateMask(L.radd).rename('radd'), ha.updateMask(L.s1).rename('s1'), ha.updateMask(L.mineria).rename('mineria'), ha.updateMask(L.urbano).rename('urbano'), ha.updateMask(L.idx.gte(1)).rename('idx1'), ha.updateMask(L.idx.gte(2)).rename('idx2') ]); var s = img.reduceRegion({ reducer: ee.Reducer.sum(), geometry: geom, scale: 30, maxPixels: 1e10, bestEffort: true }); return f.set(s); } // Semáforo de alerta a partir de los porcentajes function nivelAlerta(d) { var p1 = 100 * d.idx1 / d.total; var p2 = 100 * d.idx2 / d.total; if (p2 > SEMAFORO.altoIdx2) { return {txt: 'ALTO', color: '#d73027'}; } if (p1 > SEMAFORO.medioIdx1) { return {txt: 'MEDIO', color: '#fdae61'}; } return {txt: 'BAJO', color: '#1a9850'}; } // Gráfico: pérdida anual de bosque Hansen (ha) en el sitio function graficoHansen(geom, p) { var img = ee.Image.pixelArea().divide(10000) .updateMask(tc2000.gte(p.tc)) .addBands(lossyear.updateMask(lossyear.gt(0))); var g = img.reduceRegion({ reducer: ee.Reducer.sum().group({groupField: 1, groupName: 'anio'}), geometry: geom, scale: 30, maxPixels: 1e10, bestEffort: true }); var fc = ee.FeatureCollection(ee.List(g.get('groups')).map(function (d) { d = ee.Dictionary(d); return ee.Feature(null, {anio: ee.Number(d.get('anio')).add(2000).format('%d'), ha: d.get('sum')}); })); return ui.Chart.feature.byFeature(fc, 'anio', 'ha') .setChartType('ColumnChart') .setOptions({title: 'Pérdida anual de bosque (Hansen, ha)', legend: {position: 'none'}, colors: ['BD0026'], hAxis: {title: 'Año'}, vAxis: {title: 'ha'}}); } // Gráfico: serie de retrodispersión VH (dB) de los últimos 24 meses function graficoS1(geom, p) { var col = coleccionS1(geom) .filterDate(ee.Date(p.fin).advance(-24, 'month'), p.fin); return ui.Chart.image.series(col, geom, ee.Reducer.mean(), 200) .setOptions({title: 'Sentinel-1 VH (dB) – media del área, últimos 24 meses', vAxis: {title: 'dB'}, lineWidth: 1, pointSize: 2, colors: ['2166AC'], legend: {position: 'none'}}); } // Exportación periódica (solo funciona desde el Code Editor, no en la App publicada) // Ejecute exportarCSV() en la consola cada mes y lance la tarea en la pestaña Tasks. function exportarCSV() { var p = params(); var fc = ee.FeatureCollection(Object.keys(SITIOS).map(function (n) { return statsSitio(feat(n), p).set({ini: p.ini, fin: p.fin}); })); Export.table.toDrive({ collection: fc, description: 'Reporte_amenazas_patrimonio_' + p.fin, fileFormat: 'CSV' }); } // ============================== PASO 5 · INTERFAZ =========================== var mapa = ui.Map(); mapa.setOptions('HYBRID'); mapa.setControlVisibility({layerList: true, zoomControl: true}); var panel = ui.Panel({style: {width: '410px', padding: '8px'}}); ui.root.clear(); ui.root.add(ui.SplitPanel(panel, mapa)); var legPanel = ui.Panel({style: {position: 'bottom-left', padding: '6px 8px'}}); mapa.add(legPanel); // ---- Encabezado ---- panel.add(ui.Label('Monitoreo de amenazas al patrimonio arqueológico', { fontWeight: 'bold', fontSize: '20px', color: '#7a1f1f', margin: '4px 0'})); panel.add(ui.Label('Dashboard 2 · Alertas sobre áreas arqueológicas protegidas en Colombia', { fontSize: '12px', color: '#555'})); panel.add(ui.Label( 'EJERCICIO DE CLASE · Ingeniería Computacional\n' + 'Especialización en Sistemas de Información Geográfica\n' + 'Autor: Andrea Serna', { whiteSpace: 'pre', fontSize: '11px', color: '#333', backgroundColor: '#f3ece2', padding: '6px', margin: '6px 0'})); // ---- Controles ---- panel.add(ui.Label('1. Sitio piloto', {fontWeight: 'bold'})); var selSitio = ui.Select({items: Object.keys(SITIOS), value: Object.keys(SITIOS)[0], onChange: actualizar, style: {stretch: 'horizontal'}}); panel.add(selSitio); var hoy = new Date(); var haceN = new Date(hoy.getTime() - CONFIG.DIAS_VENTANA * 86400000); panel.add(ui.Label('2. Ventana de análisis (AAAA-MM-DD)', {fontWeight: 'bold'})); var tIni = ui.Textbox({value: fmtFecha(haceN), style: {width: '120px'}}); var tFin = ui.Textbox({value: fmtFecha(hoy), style: {width: '120px'}}); panel.add(ui.Panel([ui.Label('Inicio'), tIni, ui.Label('Fin'), tFin], ui.Panel.Layout.flow('horizontal'))); panel.add(ui.Label('3. Parámetros', {fontWeight: 'bold'})); var sUmbral = ui.Slider({min: 1, max: 6, value: 3, step: 0.5, style: {stretch: 'horizontal'}}); var sAnio = ui.Slider({min: 2015, max: 2023, value: 2020, step: 1, style: {stretch: 'horizontal'}}); var sTC = ui.Slider({min: 10, max: 90, value: 30, step: 5, style: {stretch: 'horizontal'}}); panel.add(ui.Label('Umbral de caída Sentinel-1 VH (dB)')); panel.add(sUmbral); panel.add(ui.Label('Pérdida Hansen desde el año')); panel.add(sAnio); panel.add(ui.Label('Cobertura arbórea mínima 2000 (%)')); panel.add(sTC); panel.add(ui.Label('4. Capas visibles', {fontWeight: 'bold'})); var chkIdx = ui.Checkbox('Índice de presión (0–5)', true); var chkHansen = ui.Checkbox('Deforestación Hansen (año de pérdida)', false); var chkRadd = ui.Checkbox('Alertas RADD (radar)', true); var chkGlad = ui.Checkbox('Alertas GLAD-L (Landsat)', false); var chkS1 = ui.Checkbox('Cambio Sentinel-1 VH (dB)', false); var chkMB = ui.Checkbox('MapBiomas Colombia (cobertura)', false); var chkVias = ui.Checkbox('Vías (GRIP4, requiere asset)', false); var repPanel = ui.Panel({style: {margin: '8px 0 0 0'}}); panel.add(repPanel); panel.add(ui.Label( 'Fuentes: Hansen/UMD/Google/USGS/NASA · RADD (WUR) · GLAD (UMD) · ' + 'Copernicus Sentinel-1 · MapBiomas Colombia · GRIP4.\n' + 'Herramienta de alerta temprana: verificar en campo y con imágenes de alta resolución.', {whiteSpace: 'pre-wrap', fontSize: '10px', color: '#777', margin: '10px 0 0 0'})); // ---- Lectura de parámetros ---- function params() { return {ini: tIni.getValue(), fin: tFin.getValue(), umbral: sUmbral.getValue(), anio: sAnio.getValue(), tc: sTC.getValue()}; } // ---- Leyendas (escalas de color) ---- function leyendaGradiente(titulo, paleta, txtMin, txtMax) { var barra = ui.Thumbnail({ image: ee.Image.pixelLonLat().select('longitude'), params: {bbox: [0, 0, 100, 8], dimensions: '200x16', format: 'png', min: 0, max: 100, palette: paleta}, style: {stretch: 'horizontal', margin: '0 4px', maxHeight: '18px'}}); } function leyendaCategorias(titulo, nombres, colores) { var filas = [ui.Label(titulo, {fontWeight: 'bold', fontSize: '11px', margin: '2px 4px'})]; nombres.forEach(function (n, i) { filas.push(ui.Panel([ ui.Label('', {backgroundColor: '#' + colores[i], padding: '7px', margin: '1px 4px'}), ui.Label(n, {fontSize: '10px', margin: '2px 0'}) ], ui.Panel.Layout.flow('horizontal'))); }); return ui.Panel(filas, null, {margin: '4px 0'}); } // ---- Actualizar el mapa ---- function actualizar() { var p = params(); var f = feat(selSitio.getValue()); var geom = f.geometry(); var L = capas(geom, p); } if (chkMB.getValue()) { legPanel.add(leyendaCategorias('MapBiomas ' + CONFIG.MB_ANIO, MB.nombres, MB.colores)); } } // ---- Reporte del sitio seleccionado ---- function generarReporte() { var p = params(); Revise fechas y assets.', {color: 'red'})); return; } var nv = nivelAlerta(d); function fila(txt, ha) { var pct = (100 * ha / d.total).toFixed(2); return ui.Label(txt + ': ' + ha.toFixed(1) + ' ha (' + pct + ' %)', {fontSize: '12px', margin: '1px 0'}); }); }); } // Inicio actualizar(); /***************************************************************************** * EJERCICIOS PROPUESTOS PARA LA CLASE * 1.
